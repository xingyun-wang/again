```python
"""review4 SQLite minimal 复现 P1-1-1 fix 有效性。

review3 §1.5 line 93-110 复现：
- sessionmaker(bind=engine, autoflush=False, autocommit=False, expire_on_commit=False)
- SQLite in-memory + FK enforcement
- 模拟等分 fallback 路径 db.add(ch) → KnowledgeReview(chapter_id=ch.id) → commit
"""
import os, sys, tempfile
from sqlalchemy import create_engine, event
from sqlalchemy.orm import sessionmaker
from sqlalchemy.pool import StaticPool

# FK enforcement (review3 line 178 引用 conftest.py:106-117)
engine = create_engine(
    "sqlite:///:memory:",
    connect_args={"check_same_thread": False},
    poolclass=StaticPool,
    future=True,
)

@event.listens_for(engine, "connect")
def _enable_fk(dbapi_conn, _conn_record):
    cur = dbapi_conn.cursor()
    cur.execute("PRAGMA foreign_keys=ON")
    cur.close()

# Import app models + service（review3 line 92）
sys.path.insert(0, "/home/wsq_1/tiered-homework-platform/backend")
from app.db.base import Base
from app.models import Textbook, Chapter, KnowledgeReview, KnowledgeReviewStatus
from app.services.textbook_upload import upload_textbook_with_extraction

Base.metadata.create_all(engine)

SessionLocal = sessionmaker(
    bind=engine, autoflush=False, autocommit=False, expire_on_commit=False
)
session = SessionLocal()

# Run 等分 fallback 路径
try:
    tb, chapters = upload_textbook_with_extraction(
        db=session,
        file_path="/tmp/fake.pdf",  # 实际不会真读 PDF，service 走等分 fallback
        name="test",
        subject_id=None,
        grade_level="senior_high",
        owner_user_id=1,
    )
    print(f"SUCCESS: textbook_id={tb.id}, chapters={len(chapters)}")
    for ch in chapters:
        latest_review = max(ch.reviews, key=lambda r: r.id) if ch.reviews else None
        print(f"  ch {ch.id} title={ch.title!r} review_status={latest_review.status if latest_review else None}")
except Exception as e:
    print(f"FAIL: {type(e).__name__}: {e}")
    raise
```
